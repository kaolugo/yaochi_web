RewriteEngine On

# www無しのURLに統一
RewriteCond %{HTTP_HOST} ^www\.yaochi\.work
RewriteRule ^(.*) http://yaochi.work/$1 [R=301,L]


RewriteEngine On

# index.html無しのURLに統一
RewriteCond %{THE_REQUEST} ^.*/index.html
RewriteRule ^(.*)index.html$ http://yaochi.work/$1 [R=301,L]


RewriteEngine On

# httpをhttpsに統一
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [R,L]
